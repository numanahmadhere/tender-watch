        let allMethods = {"1":{"1":"Petty Purchase","2":"Request for Quotation","3":"Open Competitive Bidding","4":"Open Bidding with Prequalification"},"2":{"5":"Petty Purchase","6":"Request for Quotation","7":"Open Competitive Bidding","8":"Open Bidding with Prequalification"},"3":{"9":"Call for Three Quotation","10":"Expression of Interest","16":"Petty Purchase","19":"Open Competition"},"4":{"11":"Petty Purchase","12":"Request for Quotation","13":"Open Competitive Bidding","14":"Open Bidding with Prequalification"}};
        let selectedMethod = 0;

        function loadMethods(type) {
            let html = `<option value="">By Procurement Method</option>`;
            let methods = allMethods[type] || [];

            $.each(methods, function(key, value) {
                html += `<option ${key == selectedMethod ? 'selected' : ''} value="${key}">${value}</option>`;
            });

            $('select[name="procurement_method"]').html(html);
        }

        function toggleMethodFilter() {
            $('#methodFilter').toggleClass('d-none', !$('select[name="type"]').val());
        }

        $('select[name="type"]').on('change', function() {
            loadMethods($(this).val());
            toggleMethodFilter();
        });
