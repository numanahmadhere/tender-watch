    <script>
        $(document).ready(function() {
            let rowsPerPage = 10;
            let rows = $('#dTable tr');
            let totalPages = Math.ceil(rows.length / rowsPerPage);

            function showPage(page) {
                rows.hide();
                rows.slice((page - 1) * rowsPerPage, page * rowsPerPage).show();
            }

            function updatePagination() {
                $('#pagination').empty();
                for (let i = 1; i <= totalPages; i++) {
                    $('#pagination').append(`<li class='page-item'><a class='page-link' href='#'>${i}</a></li>`);
                }
                $('#pagination li:first-child a').addClass('active');

                $('#pagination a').on('click', function(e) {
                    e.preventDefault();
                    let page = $(this).text();
                    showPage(page);
                    $('#pagination a').removeClass('active');
                    $(this).addClass('active');
                });
            }

            showPage(1);
            updatePagination();
        });

        function filterStatus(status) {
            $('#dTable tr').each(function() {
                let rowStatus = $(this).data('status');
                if (status === 'all' || rowStatus === status) {
                    $(this).show();
                } else {
                    $(this).hide();
                }
            });
        }

        $('#search').on('keyup', function() {
            let value = $(this).val().toLowerCase();
            $('#dTable tr').filter(function() {
                $(this).toggle($(this).text().toLowerCase().indexOf(value) > -1);
            });
        });
    </script>

        // Global Search Input Validator
        $(document).on('input paste keypress', 'input[name="search"], #searchBar, #search', function (e) {
            const input = $(this);

            if (e.type === 'keypress') {
                const charCode = (e.which) ? e.which : e.keyCode;
                // Allow only Alphanumeric (A-Z, a-z, 0-9) and Space (32)
                if (!(charCode === 32 || (charCode >= 48 && charCode <= 57) || (charCode >= 65 && charCode <= 90) || (charCode >= 97 && charCode <= 122))) {
                    e.preventDefault();
                    return false;
                }
            }

            // Cleanup on input and paste
            setTimeout(() => {
                let val = input.val();
                if (val) {
                    // Remove special characters
                    let sanitized = val.replace(/[^a-zA-Z0-9 ]/g, "");
                    // Remove multiple spaces
                    sanitized = sanitized.replace(/\s+/g, " ");
                    
                    // Filter SQL keywords
                    const sqlKeywords = ["SELECT", "INSERT", "UPDATE", "DELETE", "DROP", "UNION", "EXEC", "TRUNCATE", "OR", "AND"];
                    sqlKeywords.forEach(keyword => {
                        const regex = new RegExp("\\b" + keyword + "\\b", "gi");
                        sanitized = sanitized.replace(regex, "");
                    });

                    if (val !== sanitized) {
                        input.val(sanitized);
                    }
                }
            }, 0);
        });

        // Ensure forms are sanitized on submission
        $(document).on('submit', 'form', function (e) {
            const searchInputs = $(this).find('input[name="search"], #searchBar, #search');
            searchInputs.each(function () {
                let val = $(this).val();
                if (val) {
                    let sanitized = val.replace(/[^a-zA-Z0-9 ]/g, "").replace(/\s+/g, " ").trim();
                    const sqlKeywords = ["SELECT", "INSERT", "UPDATE", "DELETE", "DROP", "UNION", "EXEC", "TRUNCATE", "OR", "AND"];
                    sqlKeywords.forEach(keyword => {
                        const regex = new RegExp("\\b" + keyword + "\\b", "gi");
                        sanitized = sanitized.replace(regex, "");
                    });
                    $(this).val(sanitized);
                }
            });
        });
